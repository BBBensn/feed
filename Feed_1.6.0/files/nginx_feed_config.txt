# ═══════════════════════════════════════════════════════════════
# Nginx Config für feed.bensn.me — mit Basic Auth + Shared Feed
# Datei: /etc/nginx/sites-enabled/feed.bensn.me
# ═══════════════════════════════════════════════════════════════

# Schritt 1: htpasswd-Datei erstellen (einmalig auf dem Server):
#   apt install apache2-utils
#   htpasswd -c /etc/nginx/.htpasswd benni
#   (Passwort eingeben wenn gefragt)

# Schritt 2: Diese Config einspielen, dann:
#   nginx -t && systemctl reload nginx

server {
    server_name feed.bensn.me;
    client_max_body_size 20M;

    # ── /shared — öffentlich, kein Login ──
    location = /shared {
        root /var/www/feed/public;
        try_files /shared.html =404;
    }
    location = /shared.html {
        root /var/www/feed/public;
    }

    # ── /api/feed/shared — öffentlich, kein Login ──
    location = /api/feed/shared {
        proxy_pass http://127.0.0.1:5002;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }

    # ── /api/ — geschützt ──
    location /api/ {
        auth_basic "bensn feed";
        auth_basic_user_file /etc/nginx/.htpasswd;
        proxy_pass http://127.0.0.1:5002;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }

    # ── /shared/ (uploads etc) — öffentlich ──
    location /shared/ {
        alias /var/www/shared/;
    }

    # ── /uploads/ — öffentlich (Bilder müssen ohne Login laden) ──
    location /uploads/ {
        root /var/www/feed/public;
    }

    # ── / — Hauptfeed, geschützt ──
    location / {
        auth_basic "bensn feed";
        auth_basic_user_file /etc/nginx/.htpasswd;
        root /var/www/feed/public;
        index index.html;
        try_files $uri $uri/ /index.html;
    }

    listen 443 ssl;
    ssl_certificate /etc/letsencrypt/live/feed.bensn.me/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/feed.bensn.me/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
}

server {
    if ($host = feed.bensn.me) {
        return 301 https://$host$request_uri;
    }
    listen 80;
    server_name feed.bensn.me;
    return 404;
}
